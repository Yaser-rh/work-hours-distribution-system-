' OPTIMIZED MONTHLY HOUR DISTRIBUTION MACRO (V6)
' Developed by Antigravity AI Coding Assistant
'
' Enhancements implemented:
' 1. In-Memory Array Processing: Eliminates workbook cell I/O during calculation.
' 2. Feasibility Boundary Pre-Check: Prevents Excel freezing on impossible targets.
' 3. Schedule Ergonomics Heuristics: Prefers integers and consistent shift patterns.
' 4. Real-time Status Bar Updates: Informs the user of current monthly progress.

' Constants for scheduling rules
Const MAX_CONSECUTIVE_WORK_DAYS As Long = 6
Const PROBABILITY_POTENTIAL_WORKDAY As Double = (6 / 7) ' ~85.7% chance a day is considered for work

' --- Cumulative Probabilities for Hour Groups ---
Const PROB_GROUP1 As Double = 0.50 ' (Ends at 0.50) - Group 1 (Integers)
Const PROB_GROUP2 As Double = 0.30 ' (Range 0.50 to 0.80) - Group 2 (Mid Decimals)
' Group 3 (Low/High Decimals) gets the remaining 0.20 probability

' Search boundaries
Const MAX_OVERALL_ATTEMPTS As Long = 50       ' Max times to retry the whole process from scratch
Const MAX_ADJUSTMENT_ATTEMPTS As Long = 500   ' Max adjustment steps per overall attempt
Const TARGET_ROW_START As Long = 15            ' Start row to search for target hours
Const TARGET_ROW_END As Long = 20              ' End row to search for target hours

Sub DistributeMonthlyHours_WeightedExact_Optimized_V6()
    ' --- Declare Hour Arrays Locally ---
    Dim GROUP1_HOURS As Variant ' High Probability (Integers: 2, 3, 4, 5, 6)
    Dim GROUP2_HOURS As Variant ' Medium Probability (Decimals: 3.1, 3.2, 3.5, 4.1, 4.2, 4.5, 5.1, 5.2, 5.5)
    Dim GROUP3_HOURS As Variant ' Low Probability (Decimals: 2.1, 2.2, 2.5, 6.1, 6.2, 6.5)
    Dim ALL_HOURS As Variant    ' Combined list for adjustments

    ' Declare control variables
    Dim ws As Worksheet
    Dim lastCol As Long
    Dim dataCol As Long
    Dim targetHoursCol As Long
    Dim dayNameCol As Long
    Dim dateCol As Long
    Dim startRow As Long
    Dim endRow As Long
    Dim lastRow As Long
    Dim targetMonthlyHours As Double
    Dim overallAttempts As Long
    Dim success As Boolean
    Dim hrVal As Variant
    Dim idxOffset As Long
    Dim g1Count As Long, g2Count As Long, g3Count As Long, totalCount As Long
    Dim targetRowNum As Long
    Dim targetFound As Boolean
    Dim numDays As Long
    Dim dateValues As Variant
    Dim activeDaysCount As Long
    Dim i As Long
    Dim monthName As String

    ' --- Initialize the Hour Arrays ---
    GROUP1_HOURS = Array(2, 3, 4, 5, 6)
    GROUP2_HOURS = Array(3.1, 3.2, 3.5, 4.1, 4.2, 4.5, 5.1, 5.2, 5.5)
    GROUP3_HOURS = Array(2.1, 2.2, 2.5, 6.1, 6.2, 6.5)

    ' Size and build ALL_HOURS
    If Not IsArray(GROUP1_HOURS) Or Not IsArray(GROUP2_HOURS) Or Not IsArray(GROUP3_HOURS) Then GoTo InitError
    On Error Resume Next
    g1Count = UBound(GROUP1_HOURS) - LBound(GROUP1_HOURS) + 1
    g2Count = UBound(GROUP2_HOURS) - LBound(GROUP2_HOURS) + 1
    g3Count = UBound(GROUP3_HOURS) - LBound(GROUP3_HOURS) + 1
    If Err.Number <> 0 Then GoTo InitError
    On Error GoTo 0
    If g1Count <= 0 Or g2Count <= 0 Or g3Count <= 0 Then GoTo InitError

    totalCount = g1Count + g2Count + g3Count
    ReDim ALL_HOURS(1 To totalCount)

    idxOffset = 1
    For Each hrVal In GROUP1_HOURS: ALL_HOURS(idxOffset) = hrVal: idxOffset = idxOffset + 1: Next hrVal
    For Each hrVal In GROUP2_HOURS: ALL_HOURS(idxOffset) = hrVal: idxOffset = idxOffset + 1: Next hrVal
    For Each hrVal In GROUP3_HOURS: ALL_HOURS(idxOffset) = hrVal: idxOffset = idxOffset + 1: Next hrVal

    ' Sort ALL_HOURS using QuickSort helper
    QuickSort ALL_HOURS, LBound(ALL_HOURS), UBound(ALL_HOURS)

    ' Seed the random number generator
    Randomize

    On Error GoTo ErrorHandler

    Set ws = ThisWorkbook.ActiveSheet

    ' Optimize Excel application state
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False

    startRow = 4
    lastRow = 34

    ' Find the last column containing headers in Row 3
    If ws.Cells(3, ws.Columns.Count).End(xlToLeft).Column < 3 Then
        MsgBox "No month headers found in row 3 starting from column C.", vbExclamation
        GoTo CleanupAndExit
    End If
    lastCol = ws.Cells(3, ws.Columns.Count).End(xlToLeft).Column

    ' Loop through the worksheet structure (4-column blocks)
    For dataCol = 3 To lastCol Step 4
        dayNameCol = dataCol - 2
        dateCol = dataCol - 1
        targetHoursCol = dataCol + 1
        success = False
        targetFound = False
        targetRowNum = 0

        ' Extract Month Name for Progress Update
        If Not IsEmpty(ws.Cells(3, dayNameCol).Value) Then
            If IsDate(ws.Cells(3, dayNameCol).Value) Then
                monthName = Format(ws.Cells(3, dayNameCol).Value, "mmmm yyyy")
            Else
                monthName = CStr(ws.Cells(3, dayNameCol).Value)
            End If
        Else
            monthName = "Column " & Chr(64 + dataCol)
        End If

        ' --- Validation Step 1: Find Target Monthly Hours (Rows 15-20) ---
        targetMonthlyHours = 0
        For targetRowNum = TARGET_ROW_START To TARGET_ROW_END
            If targetHoursCol <= ws.Columns.Count Then
                If IsNumeric(ws.Cells(targetRowNum, targetHoursCol).Value) And Not IsEmpty(ws.Cells(targetRowNum, targetHoursCol).Value) Then
                    If ws.Cells(targetRowNum, targetHoursCol).Value > 0 Then
                        targetMonthlyHours = ws.Cells(targetRowNum, targetHoursCol).Value
                        targetFound = True
                        Exit For
                    End If
                End If
            Else
                 Exit For
            End If
        Next targetRowNum

        If Not targetFound Then GoTo NextBlock_MainLoop

        ' Update status bar for real-time tracking
        Application.StatusBar = "Optimizing " & monthName & " (Target: " & targetMonthlyHours & " hrs)..."

        ' --- Validation Step 2: Check for Dates ---
        If IsEmpty(ws.Cells(startRow, dateCol).Value) Then GoTo NextBlock_MainLoop

        ' --- Validation Step 3: Determine Data Rows & Number of Days ---
        endRow = ws.Cells(ws.Rows.Count, dateCol).End(xlUp).Row
        If endRow < startRow Then GoTo NextBlock_MainLoop
        If endRow > lastRow Then endRow = lastRow
        numDays = endRow - startRow + 1

        ' --- Optimization: Read Dates into Memory Array to Eliminate Cell I/O ---
        If numDays = 1 Then
            ReDim dateValues(1 To 1, 1 To 1)
            dateValues(1, 1) = ws.Cells(startRow, dateCol).Value
        Else
            dateValues = ws.Range(ws.Cells(startRow, dateCol), ws.Cells(endRow, dateCol)).Value
        End If

        ' Count active days
        activeDaysCount = 0
        For i = 1 To numDays
            If Not IsEmpty(dateValues(i, 1)) Then activeDaysCount = activeDaysCount + 1
        Next i

        ' --- Validation Step 4: Mathematical Feasibility Pre-Check ---
        If targetMonthlyHours > 0 Then
            Dim maxWorkdays As Long
            Dim maxPossibleHours As Double
            Dim minPossibleHours As Double

            maxWorkdays = activeDaysCount - Int(activeDaysCount / 7)
            maxPossibleHours = maxWorkdays * 6.5 ' Maximum allowed single shift
            minPossibleHours = 2.0 ' Minimum non-zero shift

            If targetMonthlyHours < minPossibleHours Or targetMonthlyHours > maxPossibleHours Then
                MsgBox "Warning: Target of " & targetMonthlyHours & " hours in " & monthName & _
                       " is mathematically impossible." & vbCrLf & _
                       "- Max hours possible under the 6-consecutive-day rule: " & maxPossibleHours & " hrs" & vbCrLf & _
                       "- Min shift length: " & minPossibleHours & " hrs" & vbCrLf & _
                       "Skipping this month block.", vbExclamation, "Infeasible Target"
                GoTo NextBlock_MainLoop
            End If
        End If

        ' Clear the target column once in-memory
        ws.Range(ws.Cells(startRow, dataCol), ws.Cells(endRow, dataCol)).ClearContents

        ' --- Processing with In-Memory Solver Retries ---
        For overallAttempts = 1 To MAX_OVERALL_ATTEMPTS
            If ProcessSingleMonth_WeightedExact_Optimized(ws, startRow, endRow, dataCol, dateValues, targetMonthlyHours, _
                                                         ALL_HOURS, GROUP1_HOURS, GROUP2_HOURS, GROUP3_HOURS) Then
                success = True
                Exit For
            End If
        Next overallAttempts

        ' Report Failure if Necessary
        If Not success Then
             MsgBox "Failed to meet exact target for " & monthName & _
                    " after " & MAX_OVERALL_ATTEMPTS & " attempts." & vbCrLf & _
                    "Target: " & targetMonthlyHours & " hrs (found in row " & targetRowNum & ")." & vbCrLf & _
                    "Leaving the column with the last failed solver state.", vbExclamation, "Solver Timeout"
        End If

NextBlock_MainLoop:
    Next dataCol

    MsgBox "Optimized hour distribution complete!", vbInformation, "Success"

CleanupAndExit:
    ' Restore Excel application state
    On Error Resume Next
    Application.StatusBar = False
    If Application.Calculation <> xlCalculationAutomatic Then Application.Calculation = xlCalculationAutomatic
    If Not Application.ScreenUpdating Then Application.ScreenUpdating = True
    If Not Application.EnableEvents Then Application.EnableEvents = True
    Set ws = Nothing
    On Error GoTo 0
    Exit Sub

InitError:
    MsgBox "Critical Error: Could not initialize hour definition arrays.", vbCritical, "Initialization Failed"
    GoTo CleanupAndExit

ErrorHandler:
    MsgBox "An error occurred in the main macro: " & vbCrLf & Err.Description, vbCritical, "Runtime Error"
    GoTo CleanupAndExit

End Sub

' ==============================================================================
' Optimized Single Month Solver (Calculates fully in-memory)
' ==============================================================================
Function ProcessSingleMonth_WeightedExact_Optimized(ws As Worksheet, ByVal startRow As Long, ByVal endRow As Long, _
                                                  ByVal dataCol As Long, ByRef dateValues As Variant, ByVal targetMonthlyHours As Double, _
                                                  ByRef ALL_HOURS As Variant, ByRef GROUP1_HOURS As Variant, _
                                                  ByRef GROUP2_HOURS As Variant, ByRef GROUP3_HOURS As Variant) As Boolean
    
    Dim dayHours() As Double
    Dim isPotentialWorkDay() As Boolean
    Dim potentialIndices() As Long
    Dim numDays As Long
    Dim i As Long, k As Long, adjAttempt As Long
    Dim consecutivePotentialDays As Long
    Dim potentialCount As Long
    Dim currentTotalHours As Double
    Dim delta As Double
    Dim isForcedOff As Boolean
    Dim randHourValue As Double
    Dim g1LBound As Long, g1UBound As Long, g2LBound As Long, g2UBound As Long, g3LBound As Long, g3UBound As Long
    Dim randPick As Double
    Dim dayIdx As Long
    Dim adjMade As Boolean
    
    Dim assignedDays() As Long, assignedCount As Long
    Dim assignedDaysS() As Long, assignedCountS As Long
    Dim randAssignedIdx As Long, swapDayIdx As Long, h_old As Double, h_new As Variant
    Dim randAssignedIdxS As Long, swapDayIdxS As Long, h_oldS As Double, h_newS As Variant, absDelta As Double

    numDays = endRow - startRow + 1
    ReDim dayHours(1 To numDays)
    ReDim isPotentialWorkDay(1 To numDays)
    ReDim potentialIndices(1 To numDays)
    
    potentialCount = 0
    consecutivePotentialDays = 0

    On Error GoTo ProcessMonthError_Func

    ' --- Pass 1: Identify Potential Workdays in-memory ---
    For i = 1 To numDays
        dayHours(i) = 0
        isPotentialWorkDay(i) = False
        
        If IsEmpty(dateValues(i, 1)) Then GoTo SkipPotentialCheck_Func

        isForcedOff = False
        If consecutivePotentialDays >= MAX_CONSECUTIVE_WORK_DAYS Then isForcedOff = True

        If Not isForcedOff Then
            If Rnd < PROBABILITY_POTENTIAL_WORKDAY Then
                 isPotentialWorkDay(i) = True
                 potentialCount = potentialCount + 1
                 potentialIndices(potentialCount) = i
                 consecutivePotentialDays = consecutivePotentialDays + 1
            Else
                 isPotentialWorkDay(i) = False
                 consecutivePotentialDays = 0
            End If
        Else
             isPotentialWorkDay(i) = False
             consecutivePotentialDays = 0
        End If
SkipPotentialCheck_Func:
    Next i

    If potentialCount > 0 Then
        ReDim Preserve potentialIndices(1 To potentialCount)
    Else
        If Abs(targetMonthlyHours) < 0.01 Then ProcessSingleMonth_WeightedExact_Optimized = True Else ProcessSingleMonth_WeightedExact_Optimized = False
        Exit Function
    End If

    ' Shuffle potential days indices to naturally distribute time off
    Call ShuffleArray(potentialIndices)

    ' --- Pass 2: Initial Weighted Assignment ---
    On Error Resume Next
    g1LBound = LBound(GROUP1_HOURS): g1UBound = UBound(GROUP1_HOURS)
    g2LBound = LBound(GROUP2_HOURS): g2UBound = UBound(GROUP2_HOURS)
    g3LBound = LBound(GROUP3_HOURS): g3UBound = UBound(GROUP3_HOURS)
    If Err.Number <> 0 Then GoTo ProcessMonthError_Func
    On Error GoTo ProcessMonthError_Func

    currentTotalHours = 0
    For k = 1 To potentialCount
        dayIdx = potentialIndices(k)
        randPick = Rnd

        If randPick < PROB_GROUP1 Then
            randHourValue = GROUP1_HOURS(Int((g1UBound - g1LBound + 1) * Rnd + g1LBound))
        ElseIf randPick < (PROB_GROUP1 + PROB_GROUP2) Then
            randHourValue = GROUP2_HOURS(Int((g2UBound - g2LBound + 1) * Rnd + g2LBound))
        Else
            randHourValue = GROUP3_HOURS(Int((g3UBound - g3LBound + 1) * Rnd + g3LBound))
        End If

        dayHours(dayIdx) = randHourValue
        currentTotalHours = currentTotalHours + dayHours(dayIdx)
    Next k

    ' --- Pass 3: Adjustment Loop with Human Ergonomics Heuristics ---
    delta = Round(targetMonthlyHours - currentTotalHours, 1)

    For adjAttempt = 1 To MAX_ADJUSTMENT_ATTEMPTS
        If Abs(delta) < 0.01 Then Exit For

        adjMade = False

        If delta > 0 Then ' ----- Need to ADD hours -----
            ' Strategy 1: Add exact delta to an OFF day
            If IsValueInArray(delta, ALL_HOURS) Then
                For k = 1 To numDays
                    If dayHours(k) = 0 And Not IsEmpty(dateValues(k, 1)) Then
                        If Check7DayRule(dayHours, k) Then
                            dayHours(k) = delta
                            currentTotalHours = currentTotalHours + delta
                            delta = Round(targetMonthlyHours - currentTotalHours, 1)
                            adjMade = True
                            Exit For
                        End If
                    End If
                Next k
            End If

            ' Strategy 2: Swap existing hours for LARGER allowed hours
            If Not adjMade Then
                 assignedCount = 0: Erase assignedDays: ReDim assignedDays(1 To numDays)
                 For i = 1 To numDays
                     If dayHours(i) > 0 Then
                         assignedCount = assignedCount + 1
                         assignedDays(assignedCount) = i
                     End If
                 Next i

                 If assignedCount > 0 Then
                     ReDim Preserve assignedDays(1 To assignedCount)
                     randAssignedIdx = Int(Rnd * assignedCount) + 1
                     swapDayIdx = assignedDays(randAssignedIdx)
                     h_old = dayHours(swapDayIdx)
                     
                     ' NATURAL HEURISTIC: Try Group 1 (Integers) first to preserve shift aesthetics
                     For Each h_new In GROUP1_HOURS
                          If h_new > h_old And Round(h_new - h_old, 1) <= delta + 0.01 Then
                               dayHours(swapDayIdx) = CDbl(h_new)
                               currentTotalHours = currentTotalHours + (h_new - h_old)
                               delta = Round(targetMonthlyHours - currentTotalHours, 1)
                               adjMade = True
                               Exit For
                          End If
                     Next h_new
                     
                     ' Fallback to ALL_HOURS (including decimals) if no clean integer fits
                     If Not adjMade Then
                         For Each h_new In ALL_HOURS
                              If h_new > h_old And Round(h_new - h_old, 1) <= delta + 0.01 Then
                                   dayHours(swapDayIdx) = CDbl(h_new)
                                   currentTotalHours = currentTotalHours + (h_new - h_old)
                                   delta = Round(targetMonthlyHours - currentTotalHours, 1)
                                   adjMade = True
                                   Exit For
                              End If
                         Next h_new
                     End If
                 End If
            End If

        ElseIf delta < 0 Then ' ----- Need to SUBTRACT hours -----
            assignedCountS = 0: Erase assignedDaysS: ReDim assignedDaysS(1 To numDays)
            For i = 1 To numDays
                If dayHours(i) > 0 Then
                    assignedCountS = assignedCountS + 1
                    assignedDaysS(assignedCountS) = i
                End If
            Next i

            If assignedCountS > 0 Then
                ReDim Preserve assignedDaysS(1 To assignedCountS)
                randAssignedIdxS = Int(Rnd * assignedCountS) + 1
                swapDayIdxS = assignedDaysS(randAssignedIdxS)
                h_oldS = dayHours(swapDayIdxS)
                absDelta = Abs(delta)

                ' Option 1: Turn off (Swap to 0)
                If Round(h_oldS, 1) <= absDelta + 0.01 Then
                      dayHours(swapDayIdxS) = 0
                      currentTotalHours = currentTotalHours - h_oldS
                      delta = Round(targetMonthlyHours - currentTotalHours, 1)
                      adjMade = True
                End If

                ' Option 2: Swap to smaller allowed value
                If Not adjMade Then
                    ' NATURAL HEURISTIC: Try Group 1 (Integers) first
                    For i = UBound(GROUP1_HOURS) To LBound(GROUP1_HOURS) Step -1
                        h_newS = GROUP1_HOURS(i)
                        If h_newS < h_oldS And Round(h_oldS - h_newS, 1) <= absDelta + 0.01 Then
                             dayHours(swapDayIdxS) = CDbl(h_newS)
                             currentTotalHours = currentTotalHours + (h_newS - h_oldS)
                             delta = Round(targetMonthlyHours - currentTotalHours, 1)
                             adjMade = True
                             Exit For
                        End If
                    Next i
                    
                    ' Fallback to ALL_HOURS (including decimals)
                    If Not adjMade Then
                        For i = UBound(ALL_HOURS) To LBound(ALL_HOURS) Step -1
                            h_newS = ALL_HOURS(i)
                            If h_newS < h_oldS And Round(h_oldS - h_newS, 1) <= absDelta + 0.01 Then
                                 dayHours(swapDayIdxS) = CDbl(h_newS)
                                 currentTotalHours = currentTotalHours + (h_newS - h_oldS)
                                 delta = Round(targetMonthlyHours - currentTotalHours, 1)
                                 adjMade = True
                                 Exit For
                            End If
                        Next i
                    End If
                End If
            End If
        End If

        ' Break early if no progress can be made to preserve CPU cycles
        If Not adjMade And adjAttempt > MAX_ADJUSTMENT_ATTEMPTS / 2 Then Exit For
    Next adjAttempt

    ' --- Write Final Results to Worksheet in a SINGLE Operation (Massive speed improvement) ---
    Dim outputValues() As Variant
    ReDim outputValues(1 To numDays, 1 To 1)
    
    For i = 1 To numDays
        If Not IsEmpty(dateValues(i, 1)) Then
            If dayHours(i) > 0 Then
                outputValues(i, 1) = dayHours(i)
            Else
                outputValues(i, 1) = Empty
            End If
        Else
            outputValues(i, 1) = Empty
        End If
    Next i

    ws.Range(ws.Cells(startRow, dataCol), ws.Cells(endRow, dataCol)).Value = outputValues

    ' Apply display formats
    For i = 1 To numDays
        If dayHours(i) > 0 Then
            If Int(dayHours(i)) = dayHours(i) Then
                ws.Cells(startRow + i - 1, dataCol).NumberFormat = "0"
            Else
                ws.Cells(startRow + i - 1, dataCol).NumberFormat = "0.0"
            End If
        End If
    Next i

    If Abs(delta) < 0.01 Then
        ProcessSingleMonth_WeightedExact_Optimized = True
    Else
        ProcessSingleMonth_WeightedExact_Optimized = False
    End If
    Exit Function

ProcessMonthError_Func:
    Debug.Print "Error in ProcessSingleMonth_WeightedExact_Optimized: " & Err.Description
    ProcessSingleMonth_WeightedExact_Optimized = False
End Function

' Helper: Shuffle Array (Fisher-Yates)
Sub ShuffleArray(arr() As Long)
    Dim i As Long, j As Long, temp As Long, lower As Long, upper As Long
    On Error Resume Next: lower = LBound(arr): upper = UBound(arr)
    If Err.Number <> 0 Or lower >= upper Then On Error GoTo 0: Exit Sub
    On Error GoTo 0: For i = upper To lower + 1 Step -1
    j = Int((i - lower + 1) * Rnd + lower): temp = arr(i): arr(i) = arr(j): arr(j) = temp
    Next i
End Sub

' Helper: Check if value exists in array within tolerance
Function IsValueInArray(valueToFind As Variant, arr As Variant) As Boolean
    Dim element As Variant: IsValueInArray = False: If Not IsArray(arr) Then Exit Function
    On Error Resume Next: For Each element In arr
    If Abs(CDbl(element) - CDbl(valueToFind)) < 0.01 Then IsValueInArray = True: Exit Function
    Next element: On Error GoTo 0
End Function

' Helper: Check if turning checkIndex workday ON violates the 6-consecutive-day limit
Function Check7DayRule(ByRef hoursArray() As Double, checkIndex As Long) As Boolean
    Dim consecutive As Long
    Dim i As Long
    Dim lowerB As Long, upperB As Long

    On Error Resume Next
    lowerB = LBound(hoursArray): upperB = UBound(hoursArray)
    If Err.Number <> 0 Then
        Check7DayRule = False
        Exit Function
    End If
    On Error GoTo 0

    If checkIndex < lowerB Or checkIndex > upperB Then
        Check7DayRule = False
        Exit Function
    End If

    consecutive = 0
    ' Count backward from day before checkIndex
    For i = checkIndex - 1 To lowerB Step -1
         If hoursArray(i) > 0 Then consecutive = consecutive + 1 Else Exit For
    Next i

    ' Count forward from day after checkIndex
    For i = checkIndex + 1 To upperB Step 1
         If hoursArray(i) > 0 Then consecutive = consecutive + 1 Else Exit For
    Next i

    If consecutive + 1 <= MAX_CONSECUTIVE_WORK_DAYS Then
        Check7DayRule = True
    Else
        Check7DayRule = False
    End If
End Function

' Helper: QuickSort for numeric variants
Sub QuickSort(arr As Variant, L As Long, R As Long)
    Dim i As Long, j As Long
    Dim pivot As Double
    Dim temp As Variant

    On Error GoTo QuickSortError
    If L >= R Then Exit Sub

    i = L: j = R
    pivot = CDbl(arr(L + Int((R - L) / 2)))

    Do
        Do While CDbl(arr(i)) < pivot: i = i + 1: Loop
        Do While CDbl(arr(j)) > pivot: j = j - 1: Loop

        If i <= j Then
            temp = arr(i): arr(i) = arr(j): arr(j) = temp
            i = i + 1: j = j - 1
        End If
    Loop While i <= j

    If L < j Then QuickSort arr, L, j
    If i < R Then QuickSort arr, i, R
    Exit Sub
    
QuickSortError:
     Debug.Print "QuickSort Error: " & Err.Description
End Sub
